d3.csv('https://raw.githubusercontent.com/plotly/datasets/master/2014_usa_states.csv', (err, rows) => { var unpack = (rows, key) => rows.map((row) => row[key]); var data = [{ type: 'choropleth', locationmode: 'USA-states', locations: unpack(rows, 'Postal'), z: unpack(rows, 'Population'), text: unpack(rows, 'State'), autocolorscale: true }]; var layout = { title: {text: '2014 US Population by State'}, geo:{ scope: 'usa', countrycolor: 'rgb(255, 255, 255)', showland: true, landcolor: 'rgb(217, 217, 217)', showlakes: true, lakecolor: 'rgb(255, 255, 255)', subunitcolor: 'rgb(255, 255, 255)', lonaxis: {}, lataxis: {} } }; Plotly.newPlot("myDiv", data, layout); });
